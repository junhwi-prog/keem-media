/* 마이해빗 작업용: 옵션 목록 정리 v5 (2026-10-08)
   - v5: 디다 형식 줄(박스 사진·칩·고른 줄 테두리) + 할인율 유지 + 옵션 아래 혜택 줄(채널 추가 쿠폰·카카오 가입)
   - v4: 다른 상품(단품·기획) 줄을 골라도 페이지 이동 없이 이 화면에서 담기·바로구매·네이버페이 (카페24 장바구니 요청에 상품 번호·품목코드만 바꿔 보냄)
   - 단품 옵션과 기획(N+1)을 박스 수 순서로 한 목록에. 같은 박스 수는 가장 싼 줄 하나만 (값이 같으면 이 페이지에서 바로 고르는 줄)
   - 가격은 옵션별 실제 판매가(option_stock_data)로 읽는다 (옵션 이름에 "+○원"이 없는 상품이 있다)
   - BEST 는 최근 30일 많이 팔린 줄(BEST_OF), 최대할인은 박스당 가장 싼 줄. 옵션 이름에 박힌 [BEST]·[최대할인]은 숨김
   - 가격·상품명·관리자 옵션은 그대로. 숨긴 옵션도 지우지 않고 화면에서만 감춘다 */
(function () {

  // 단품 product_no -> 판매 중인 기획 product_no (2026-10-08 진열·판매 기준. 368·496·495 는 진열 안 함이라 뺌)
  var BUNDLES = {
    14: [493], 268: [371], 471: [485], 482: [494, 484], 62: [454, 239],
    207: [328], 327: [350], 365: [386], 366: [386], 411: [479],
    176: [226], 440: [450]
  };
  var SKIP = { 386: 1 };   // 맛 두 개를 고르는 기획 페이지는 스킨 화면 그대로
  var NOTE = { 386: '맛 섞기 가능' };
  // BEST: 최근 30일(9/8~10/7) 가장 많이 팔린 줄. 값 = 기획 번호, 또는 's'+박스 수(단품 옵션)
  var BEST_OF = { 14: 493, 268: '371:2+1', 471: 485, 482: 484, 62: 454, 207: 328, 327: 350, 365: 386, 366: 386, 176: 226 };
  var SINGLE_OF = {};
  Object.keys(BUNDLES).forEach(function (s) { BUNDLES[s].forEach(function (b) { if (!SINGLE_OF[b]) SINGLE_OF[b] = +s; }); });

  var PREFIX = (location.pathname.match(/^\/skin-skin\d+/) || [''])[0];
  function won(n) { return Math.round(n).toLocaleString('ko-KR') + '원'; }
  function num(s) { return parseInt(String(s).replace(/[^\d]/g, ''), 10) || 0; }
  function meta(doc, p) { var m = doc.querySelector('meta[property="' + p + '"]'); return m ? m.getAttribute('content') : ''; }
  function boxesOf(t) { var p = /(\d+)\s*\+\s*(\d+)/.exec(t); if (p) return +p[1] + +p[2]; var m = /(\d+)\s*박스/.exec(t); return m ? +m[1] : 0; }
  function plusOf(t) { var p = /(\d+)\s*\+\s*(\d+)/.exec(t); return p ? p[1] + '+' + p[2] : ''; }
  function pctOf(t) { var m = /(\d+)%/.exec(t); return m ? +m[1] : 0; }
  function productNo() { return parseInt(window.iProductNo || (/product_no=(\d+)/.exec(location.search) || [])[1], 10); }
  function wantBox() { var m = /[?&]mh_box=(\d+)/.exec(location.search); return m ? +m[1] : 0; }
  function href(no, box) { return PREFIX + '/product/detail.html?product_no=' + no + (box ? '&mh_box=' + box : ''); }

  // 옵션별 실제 판매가: { 옵션값: {text, total, show} }
  function stockOf(html) {
    var m = /option_stock_data\s*=\s*'((?:\\.|[^'\\])*)'/.exec(html || ''); if (!m) return {};
    try {
      var d = JSON.parse(JSON.parse('"' + m[1] + '"')), r = {};
      Object.keys(d).forEach(function (k) { var v = d[k]; if (v && v.option_value != null) r[k] = { text: v.option_value, total: +v.option_price, show: v.is_display === 'T' && v.is_selling === 'T' }; });
      return r;
    } catch (e) { return {}; }
  }
  // 옵션값이 품목코드가 아니라 옵션 이름 그대로인 상품이 있다 (효소습관 맥스·NMN·슬림케어)
  function byText(st, v) { for (var k in st) if (st[k].text === v) return st[k]; return null; }
  // 기획 줄 할인율은 단품 1박스 정가 × 박스 수 기준으로 맞춘다 (기획 페이지마다 정가 기준이 달라서)
  function fixPct(rows, l1) {
    var one = rows.filter(function (r) { return r.n === 1 && !r.bno; })[0];
    if (one && !one.pct) l1 = 0;   // 단품에 할인율 표시가 없는 상품(NMN)은 기획 줄에도 안 붙인다
    else if (one) l1 = one.total / (1 - one.pct / 100);   // 1박스 줄에 보이는 할인율과 같은 기준으로 (상품 정가 칸과 옵션 이름의 %가 안 맞는 상품이 있다)
    if (!l1) { rows.forEach(function (r) { if (r.bno) r.pct = 0; }); return; }
    rows.forEach(function (r) { if (r.bno && r.n) r.pct = Math.max(0, Math.round((1 - r.total / (r.n * l1)) * 100)); });
  }
  function fetchPage(no) {
    return fetch('/product/detail.html?product_no=' + no, { credentials: 'same-origin' })
      .then(function (r) { return r.ok ? r.text() : ''; })
      .then(function (h) { return h ? { no: no, html: h, doc: new DOMParser().parseFromString(h, 'text/html') } : null; })
      .catch(function () { return null; });
  }
  function priceOf(doc) { return num(meta(doc, 'product:sale_price:amount') || meta(doc, 'product:price:amount')); }
  function listOf(doc) { var cu = doc.querySelector('#span_product_price_custom'); return num(cu ? cu.textContent : '') || num(meta(doc, 'product:price:amount')); }

  // 다른 페이지의 줄을 이 화면에서 살 때 필요한 값 (상품 번호·품목코드·옵션값). 못 구하면 null → 그 줄은 링크로 남는다
  function buyOf(pg, key) {
    if (SKIP[pg.no]) return null;
    var b = { no: pg.no, name: meta(pg.doc, 'og:title') || '', price: priceOf(pg.doc) };
    if (!key) {
      var m = /item_code\s*=\s*'([^']+)'/.exec(pg.html || ''); if (!m) return null;
      b.code = m[1]; b.opt = ''; return b;
    }
    var st = stockOf(pg.html), o = st[key], v = '';
    [].some.call(pg.doc.querySelectorAll('#product_option_id1 option'), function (x) {
      if (x.value === key || (o && (x.value === o.text || x.textContent.trim() === o.text))) { v = x.value; return true; }
    });
    if (!v) return null;
    b.code = key; b.opt = v; return b;
  }
  // 다른 페이지의 줄들
  function rowsOfPage(pg, isSingle) {
    if (!pg) return [];
    var st = stockOf(pg.html), keys = Object.keys(st), title = meta(pg.doc, 'og:title') || '', rows = [];
    if (isSingle) {
      if (!keys.length) rows.push({ n: 1, total: priceOf(pg.doc), pct: pctFromList(pg.doc), href: href(pg.no), src: 's1', buy: buyOf(pg, '') });

      if (plusKeys.length) plusKeys.forEach(function (k) { var o = st[k], r = bundleRow(pg, o.text, o.total); r.buy = buyOf(pg, k); rows.push(r); });
      else if (plusOf(title)) { var r = bundleRow(pg, title, priceOf(pg.doc)); r.buy = buyOf(pg, ''); rows.push(r); }
    }
    return rows;
  }
  function pctFromList(doc) { var l = listOf(doc), s = priceOf(doc); return l > s ? Math.round((1 - s / l) * 100) : 0; }
  function bundleRow(pg, text, total) {
    var n = boxesOf(text), plus = plusOf(text), notes = [];
    if (NOTE[pg.no]) notes.push(NOTE[pg.no]);
    var list = listOf(pg.doc) * (total / (priceOf(pg.doc) || total));
    return { n: n, total: total, plus: plus, notes: notes, pct: list > total ? Math.round((1 - total / list) * 100) : 0, href: href(pg.no), src: pg.no + ':' + plus, bno: pg.no };
  }

  // 같은 박스 수 중 가장 싼 줄 하나. 값이 같으면 이 페이지 줄(here) > 단품 > 기획
  function pick(rows) {
    var by = {}, out = [];
    rows.forEach(function (r) {
      if (!r.n) { out.push(r); return; }
      var c = by[r.n];
      var rank = function (x) { return (x.here ? 0 : 2) + (x.src && x.src.charAt(0) === 's' ? 0 : 1); };
      if (!c || r.total < c.total || (r.total === c.total && rank(r) < rank(c))) by[r.n] = r;
    });
    Object.keys(by).forEach(function (n) { out.push(by[n]); });
    return out.sort(function (a, b) { return (a.n || 0) - (b.n || 0); });
  }

  // v5 (디다 형식, 10/8 사용자 "좋은 점만"): 왼쪽 박스 사진(실제 누끼를 박스 수만큼), 이름 아래 총액 + 할인율(빨강) + 박스당(회색), 오른쪽 끝 칩
  var THUMB = { 14: 'enzyme', 268: 'glucose', 471: 'mct', 482: 'stopit', 62: 'iron', 207: 'vinegar', 327: 'lemon', 365: 'max', 366: 'max_gb', 411: 'nmn', 176: 'slim', 440: 'wildfit' };   // 디자인팀 누끼로 만든 썸네일 (thumbs/compose.py)   // 단품 번호 -> 썸네일 이름 (이름_N.png). 사진 없는 상품은 사진 칸 없이

  var THUMB_BASE = window.MH_THUMB_BASE || 'https://cdn.jsdelivr.net/gh/junhwi-prog/keem-media@7f8970e/myhabit/option_thumbs/';

  function rowHTML(o) {

    var th = THUMB[CUR_SINGLE] && o.n ? '<span class="mh-th"><img src="' + THUMB_BASE + THUMB[CUR_SINGLE] + '_' + Math.min(o.n, 6) + '.png" alt=""></span>' : '';

      (o.total >= FREE_SHIP ? '<em class="mh-chip">무료배송</em>' : '');
    var notes = (o.notes || []).length ? '<span class="mh-note">' + o.notes.join(' · ') + '</span>' : '';
    var pb = o.n ? '<span class="mh-pb">박스당 ' + won(o.total / o.n) + '</span>' : '';

      '<span class="mh-pr"><span class="mh-won">' + won(o.total) + '</span>' + (o.pct ? '<span class="mh-pct">' + o.pct + '%</span>' : '') + pb + '</span>' + notes + '</span>' +

  }

  function hideOldBanner() {

  function markTags(rows, single) {

  function foreignLi(r) {

  function render(ul, rows) {

  /* ---------- 옵션 목록이 있는 페이지 (단품, 또는 옵션 있는 기획) ---------- */
  function decorateList() {

      var r = { li: li, here: true, n: n, total: o ? o.total : base, plus: plus, pct: pctOf(t), src: isSingle ? 's' + n : no + ':' + plus, bno: isSingle ? 0 : no, notes: [] };

    var others = (BUNDLES[single] || []).filter(function (b) { return b !== no; });

  function buildPlain() {

    if (window.__mhPlain || !single || SKIP[no] || !anchor || document.querySelector('.mh-list') || document.querySelector('#product_option_id1')) return;

    var isSingle = single === no; CUR_SINGLE = single;
    var me = { no: no, html: document.documentElement.innerHTML, doc: document };
    var mine = rowsOfPage(me, isSingle); mine.forEach(function (r) { r.here = true; r.on = true; delete r.href; });
    var others = (BUNDLES[single] || []).filter(function (b) { return b !== no; });
    var l1 = isSingle ? listOf(document) : 0;
    var jobs = (isSingle ? [] : [fetchPage(single).then(function (pg) { if (pg) { l1 = listOf(pg.doc); window.__mhName = meta(pg.doc, "og:title"); } return rowsOfPage(pg, true); })])
